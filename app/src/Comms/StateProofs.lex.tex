definition{
  name: capped,
  parameters: [limit : Nat, value : Nat],
  result: Nat,
  body: (if (limit <? value) then limit else value)
}

definition{
  name: rawStatus,
  parameters: [
    member : Nat,
    creator : Nat,
    creatorPower : Nat,
    required : Nat,
    open : Nat,
    events : (List (List Nat))
  ],
  result: Nat,
  body: (match events
    | List.nil => (if (member =? creator) then 2 else 0)
    | List.cons head rest => (Comms.State::step 0 (member =? (Comms.State::subjectOf head)) (Comms.State::actionOf head) (Comms.State::state 0 (Comms.State::senderOf head) creator creatorPower required open rest) (Comms.State::state 1 (Comms.State::senderOf head) creator creatorPower required open rest) (capped 4 (rawStatus member creator creatorPower required open rest)) (Comms.State::state 1 member creator creatorPower required open rest) required open (Comms.State::grantedOf head))),
  recursive_argument: events
}

definition{
  name: rawPower,
  parameters: [
    member : Nat,
    creator : Nat,
    creatorPower : Nat,
    required : Nat,
    open : Nat,
    events : (List (List Nat))
  ],
  result: Nat,
  body: (match events
    | List.nil => (if (member =? creator) then creatorPower else 0)
    | List.cons head rest => (Comms.State::step 1 (member =? (Comms.State::subjectOf head)) (Comms.State::actionOf head) (Comms.State::state 0 (Comms.State::senderOf head) creator creatorPower required open rest) (capped creatorPower (rawPower (Comms.State::senderOf head) creator creatorPower required open rest)) (Comms.State::state 0 member creator creatorPower required open rest) (capped creatorPower (rawPower member creator creatorPower required open rest)) required open (Comms.State::grantedOf head))),
  recursive_argument: events
}

  proof: linear_arithmetic{definitions: [Comms.State::state]},
  axioms: [propext]
}

theorem{
  name: statusIsCapped,
  parameters: [
    member : Nat,
    creator : Nat,
    creatorPower : Nat,
    required : Nat,
    open : Nat,
    events : (List (List Nat))
  ],
  statement: ((Comms.State::state 0 member creator creatorPower required open events) = (capped 4 (rawStatus member creator creatorPower required open events))),
  proof: induction{
    branches: [
      {
        proof: linear_arithmetic{definitions: [Comms.State::state, capped, rawStatus]},
        binders: [],
        constructor: nil
      },
      {
        proof: simplify{definitions: [cappedStatusStaysCapped, ih, rawStatus, Comms.State::state]},
        binders: [head, rest, ih],
        constructor: cons
      }
    ],
    scrutinee: events
  },
  axioms: [Quot.sound, propext]
}
