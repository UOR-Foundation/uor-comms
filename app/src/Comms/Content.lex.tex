definition{
  name: fieldsAllowed,
  parameters: [
    version : Nat,
    kind : Nat,
    relates : Nat,
    forwarded : Nat,
    everyone : Nat,
    mentions : Nat,
    textLength : Nat,
    attachments : Nat,
    keyLength : Nat

  result: Bool,
  body: ((version =? 1) && ((relates <=? 1) && (((kind =? 3) && ((relates =? 1) && ((1 <=? keyLength) && (keyLength <=? 32)))) || (((kind =? 1) || ((kind =? 2) && (relates =? 1))) && ((forwarded <=? 1) && ((everyone <=? 1) && ((mentions <=? 50) && ((1 <=? textLength) && ((textLength <=? 4096) && (attachments =? 0))))))))))
}

definition{
  name: byteOf,
  parameters: [value : Bytes, at : Nat],
  result: Nat,
  body: (match (Foundation.Bytes::byteAt value at)
    | Option.none => 256
    | Option.some byte => (prim to_nat byte : Nat)),
  axioms: [Quot.sound, propext]
}

definition{
  name: numberOf,
  parameters: [value : Bytes, at : Nat, width : Nat, gathered : Nat],
  result: Nat,
  body: (match width
    | Nat.zero => gathered
    | Nat.succ fewer => (numberOf value (at + 1) fewer ((prim multiply gathered 256 : Nat) + (byteOf value at)))),
  recursive_argument: width
}

definition{
  name: textClean,
  parameters: [value : Bytes, start : Nat, remaining : Nat],
  result: Bool,
  body: (match remaining
    | Nat.zero => true
    | Nat.succ rest => ((not (((byteOf value (start + rest)) =? 0) || ((byteOf value (start + rest)) =? 13))) && (textClean value start rest))),
  recursive_argument: remaining
}

definition{
  name: keyClean,
  parameters: [value : Bytes, start : Nat, remaining : Nat],
  result: Bool,
  body: (match remaining
    | Nat.zero => true
    | Nat.succ rest => ((not (((byteOf value (start + rest)) =? 0) || (((byteOf value (start + rest)) =? 9) || (((byteOf value (start + rest)) =? 10) || (((byteOf value (start + rest)) =? 13) || ((byteOf value (start + rest)) =? 32)))))) && (keyClean value start rest))),
  recursive_argument: remaining
}

definition{
  name: wordsBefore,
  parameters: [value : Bytes, left : Nat, right : Nat, remaining : Nat],
  result: Bool,
  body: (match remaining
    | Nat.zero => false
    | Nat.succ rest => (if ((numberOf value left 8 0) <? (numberOf value right 8 0)) then true else (if ((numberOf value left 8 0) =? (numberOf value right 8 0)) then (wordsBefore value (left + 8) (right + 8) rest) else false))),
  recursive_argument: remaining
}

definition{
  name: mentionsAscend,
  parameters: [value : Bytes, start : Nat, remaining : Nat],
  result: Bool,
  body: (match remaining
    | Nat.zero => true
    | Nat.succ rest => (if (wordsBefore value (start + (prim multiply 32 rest : Nat)) ((start + (prim multiply 32 rest : Nat)) + 32) 4) then (mentionsAscend value start rest) else false)),
  recursive_argument: remaining
}

definition{
  name: headerAt,
  parameters: [value : Bytes, base : Nat],
  result: Nat,
  body: (11 + (if ((byteOf value (base + 10)) =? 1) then 32 else 0))
}

definition{
  name: textAtIn,
  parameters: [value : Bytes, base : Nat],
  result: Nat,
  body: (((base + (headerAt value base)) + 3) + (prim multiply 32 (byteOf value ((base + (headerAt value base)) + 2)) : Nat))
}

definition{
  name: textLengthIn,
  parameters: [value : Bytes, base : Nat],
  result: Nat,
  body: (numberOf value (textAtIn value base) 4 0)
}

  parameters: [value : Bytes, base : Nat, length : Nat],

(prim subtract (((textAtIn value base) + 5) + (textLengthIn value base)) base : Nat)) && ((textClean value ((textAtIn value base) + 4) (textLengthIn value base)) && (mentionsAscend value ((base + (headerAt value base)) + 3) (prim subtract (byteOf value ((base + (headerAt value base)) + 2)) 1 : Nat))))))

definition{
  name: kindIn,
  parameters: [value : Bytes, base : Nat, length : Nat],
  result: Nat,
  body: (if (validIn value base length) then (byteOf value (base + 1)) else 0)
}
