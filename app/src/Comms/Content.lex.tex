definition{
  name: bigEndian,
  parameters: [bytes : (List Nat), start : Nat, width : Nat, gathered : Nat],
  result: Nat,
  body: (match width
    | Nat.zero => gathered
    | Nat.succ fewer => (bigEndian bytes (start + 1) fewer ((prim multiply gathered 256 : Nat) + (Comms.Event::wordAt bytes start)))),
  recursive_argument: width
}

definition{
  name: badTextBytes,
  parameters: [text : (List Nat)],
  result: Nat,
  body: (match text
    | List.nil => 0
    | List.cons head rest => ((if ((head =? 0) || (head =? 13)) then 1 else 0) + (badTextBytes rest))),
  recursive_argument: text
}

definition{
  name: badKeyBytes,
  parameters: [key : (List Nat)],
  result: Nat,
  body: (match key
    | List.nil => 0
    | List.cons head rest => ((if ((head =? 0) || ((head =? 9) || ((head =? 10) || ((head =? 13) || (head =? 32))))) then 1 else 0) + (badKeyBytes rest))),
  recursive_argument: key
}

definition{
  name: addressWords,
  parameters: [bytes : (List Nat), start : Nat, words : Nat],
  result: Comms.Event::Words,
  body: (match words
    | Nat.zero => (record Comms.Event::Words {items := (nil Nat)})
    | Nat.succ fewer => (record Comms.Event::Words {items := (cons (bigEndian bytes start 8 0) (project (addressWords bytes (start + 8) fewer) items))})),
  recursive_argument: words
}

definition{
  name: slice,
  parameters: [bytes : (List Nat), start : Nat, length : Nat],
  result: Comms.Event::Words,
  body: (Comms.Event::take (project (Comms.Event::skip bytes start) items) length)
}

}
