\title{Boolean}
\begin{semanticmodule}
definition{name: idle, parameters: [], result: Nat, body: 0}

definition{name: calling, parameters: [], result: Nat, body: 1}

definition{name: ringing, parameters: [], result: Nat, body: 2}

definition{name: connected, parameters: [], result: Nat, body: 3}

definition{name: ended, parameters: [], result: Nat, body: 4}

definition{
  name: step,
  parameters: [state : Nat, signal : Nat],
  result: Nat,
  body: (if (4 <=? state) then 4 else (if (signal =? 4) then 4 else (if (state =? 0) then (if (signal =? 0) then 1 else (if (signal =? 1) then 2 else 0)) else (if (state =? 1) then (if (signal =? 3) then 3 else 1) else (if (state =? 2) then (if (signal =? 2) then 3 else 2) else 3)))))
}

definition{
  name: yields,
  parameters: [ownOffer : Nat, otherOffer : Nat],
  result: Bool,
  body: (ownOffer <? otherOffer)
}
\end{semanticmodule}
\end{lexlean}
