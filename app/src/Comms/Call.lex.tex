\begin{lexlean}{Comms.Call}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\title{Boolean}
\begin{semanticmodule}
definition{
  name: idle,
  parameters: [],
  result: Nat,
  body: 0
}

definition{
  name: calling,
  parameters: [],
  result: Nat,
  body: 1
}

definition{
  name: ringing,
  parameters: [],
  result: Nat,
  body: 2
}

definition{
  name: connected,
  parameters: [],
  result: Nat,
  body: 3
}

definition{
  name: ended,
  parameters: [],
  result: Nat,
  body: 4
}

definition{
  name: step,
  parameters: [state : Nat, signal : Nat],
  result: Nat,
  body: (if (4 <=? state) then 4 else (if (signal =? 4) then 4 else (if (state =? 0) then (if (signal =? 0) then 1 else (if (signal =? 1) then 2 else 0)) else (if (state =? 1) then (if (signal =? 3) then 3 else 1) else (if (state =? 2) then (if (signal =? 2) then 3 else 2) else 3)))))
}

definition{
  name: yields,
  parameters: [ownOffer : Nat, otherOffer : Nat],
  result: Bool,
  body: (ownOffer <? otherOffer)
}

theorem{
  name: endedCallStaysEnded,
  parameters: [signal : Nat],
  statement: ((step 4 signal) = 4),
  proof: linear_arithmetic{definitions: [step]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: hangUpAlwaysEnds,
  parameters: [state : Nat],
  statement: ((step state 4) = 4),
  proof: linear_arithmetic{definitions: [step]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: noConnectionFromIdle,
  parameters: [signal : Nat],
  statement: (((step 0 signal) =? 3) = false),
  proof: linear_arithmetic{definitions: [step]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: callerConnectsOnlyWhenAnswered,
  parameters: [signal : Nat],
  statement: ((((step 1 signal) =? 3) && ((signal <? 3) || (3 <? signal))) = false),
  proof: linear_arithmetic{definitions: [step]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: calleeConnectsOnlyByAccepting,
  parameters: [signal : Nat],
  statement: ((((step 2 signal) =? 3) && ((signal <? 2) || (2 <? signal))) = false),
  proof: linear_arithmetic{definitions: [step]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: stateStaysInRange,
  parameters: [state : Nat, signal : Nat],
  statement: (((step state signal) <=? 4) = true),
  proof: linear_arithmetic{definitions: [step]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: crossedCallsNeverBothYield,
  parameters: [ownOffer : Nat, otherOffer : Nat],
  statement: (((yields ownOffer otherOffer) && (yields otherOffer ownOffer)) = false),
  proof: linear_arithmetic{definitions: [yields]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: crossedCallsOneSideYields,
  parameters: [ownOffer : Nat, otherOffer : Nat],
  statement: (((ownOffer =? otherOffer) || ((yields ownOffer otherOffer) || (yields otherOffer ownOffer))) = true),
  proof: linear_arithmetic{definitions: [yields]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: namedStatesMatch,
  parameters: [],
  statement: ((((idle) =? 0) && (((calling) =? 1) && (((ringing) =? 2) && (((connected) =? 3) && ((ended) =? 4))))) = true),
  proof: reflexivity{},
  axioms: []
}
\end{semanticmodule}
\end{lexlean}
