  name: arrivalPositionChangesNothing,

  proof: induction{
    branches: [
      {
        proof: simplify{definitions: [Comms.Resolve::joined, Comms.Resolve::resolve]},
        binders: [],
        constructor: nil

      {
        proof: simplify{
          definitions: [ih, mergeLeftComm, Comms.Resolve::joined, Comms.Resolve::resolve]

        constructor: cons
      }

  },
  axioms: [Quot.sound, propext]
