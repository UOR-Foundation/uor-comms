theorem{
  name: arrivalPositionChangesNothing,
  parameters: [before : (List Nat), event : Nat, after : (List Nat), seed : Nat],
  statement: ((Comms.Resolve::resolve (Comms.Resolve::joined before (cons event after)) seed) = (Comms.Resolve::merge event (Comms.Resolve::resolve (Comms.Resolve::joined before after) seed))),
  proof: induction{
    branches: [
      {proof: simplify{definitions: [Comms.Resolve::joined, Comms.Resolve::resolve]}, binders: [], constructor: nil},
      {
        proof: simplify{definitions: [ih, mergeLeftComm, Comms.Resolve::joined, Comms.Resolve::resolve]},
        binders: [head, rest, ih],
        constructor: cons
      }
    ],
    scrutinee: before
  },
  axioms: [Quot.sound, propext]
}

  },
