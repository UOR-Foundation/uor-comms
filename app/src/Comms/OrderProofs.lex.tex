\useglossary{lexlean.std.nat@1.2.0}
\importmodule{Comms.Order}
\title{Boolean}
\begin{semanticmodule}
theorem{
  name: orderIsTotal,
  parameters: [leftRank : Nat, leftKappa : Nat, rightRank : Nat, rightKappa : Nat],
  statement: (((Comms.Order::precedes leftRank leftKappa rightRank rightKappa) || ((Comms.Order::precedes rightRank rightKappa leftRank leftKappa) || (Comms.Order::same leftRank leftKappa rightRank rightKappa))) = true),
  proof: linear_arithmetic{definitions: [Comms.Order::precedes, Comms.Order::same]},
  axioms: [Classical.choice, Quot.sound, propext]
}

theorem{
  name: orderNeverBothWays,
  parameters: [leftRank : Nat, leftKappa : Nat, rightRank : Nat, rightKappa : Nat],
  statement: (((Comms.Order::precedes leftRank leftKappa rightRank rightKappa) && (Comms.Order::precedes rightRank rightKappa leftRank leftKappa)) = false),
  proof: linear_arithmetic{definitions: [Comms.Order::precedes]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: orderNeverPrecedesItself,
  parameters: [rank : Nat, kappa : Nat],
  statement: ((Comms.Order::precedes rank kappa rank kappa) = false),
  proof: linear_arithmetic{definitions: [Comms.Order::precedes]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: orderIsTransitive,
  parameters: [aRank : Nat, aKappa : Nat, bRank : Nat, bKappa : Nat, cRank : Nat, cKappa : Nat],
  statement: (((Comms.Order::precedes aRank aKappa bRank bKappa) && ((Comms.Order::precedes bRank bKappa cRank cKappa) && ((Comms.Order::precedes cRank cKappa aRank aKappa) || (Comms.Order::same aRank aKappa cRank cKappa)))) = false),
  proof: linear_arithmetic{definitions: [Comms.Order::precedes, Comms.Order::same]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: earlierInOrderHasSmallerKey,
  parameters: [leftRank : Nat, leftKappa : Nat, rightRank : Nat, rightKappa : Nat],
  statement: ((((leftKappa <? (Comms.Order::addressBound)) && (rightKappa <? (Comms.Order::addressBound))) && ((Comms.Order::precedes leftRank leftKappa rightRank rightKappa) && ((Comms.Order::orderingKey rightRank rightKappa) <=? (Comms.Order::orderingKey leftRank leftKappa)))) = false),
  proof: linear_arithmetic{
    definitions: [Comms.Order::addressBound, Comms.Order::orderingKey, Comms.Order::precedes]
  },
  axioms: [Quot.sound, propext]
}

theorem{
  name: smallerKeyIsEarlierInOrder,
  parameters: [leftRank : Nat, leftKappa : Nat, rightRank : Nat, rightKappa : Nat],
  statement: ((((leftKappa <? (Comms.Order::addressBound)) && (rightKappa <? (Comms.Order::addressBound))) && (((Comms.Order::orderingKey leftRank leftKappa) <? (Comms.Order::orderingKey rightRank rightKappa)) && ((Comms.Order::precedes rightRank rightKappa leftRank leftKappa) || (Comms.Order::same leftRank leftKappa rightRank rightKappa)))) = false),
  proof: linear_arithmetic{
    definitions: [
      Comms.Order::addressBound,
      Comms.Order::orderingKey,
      Comms.Order::precedes,
      Comms.Order::same
    ]
  },
  axioms: [Quot.sound, propext]
}

theorem{
  name: equalKeysAreTheSameEvent,
  parameters: [leftRank : Nat, leftKappa : Nat, rightRank : Nat, rightKappa : Nat],
  statement: ((((leftKappa <? (Comms.Order::addressBound)) && (rightKappa <? (Comms.Order::addressBound))) && (((Comms.Order::orderingKey leftRank leftKappa) =? (Comms.Order::orderingKey rightRank rightKappa)) && ((Comms.Order::precedes leftRank leftKappa rightRank rightKappa) || (Comms.Order::precedes rightRank rightKappa leftRank leftKappa)))) = false),
  proof: linear_arithmetic{
    definitions: [Comms.Order::addressBound, Comms.Order::orderingKey, Comms.Order::precedes]
  },
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
