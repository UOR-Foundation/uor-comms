\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\title{Boolean}
\begin{semanticmodule}
definition{
  name: nextStatus,
  parameters: [action : Nat, targetStatus : Nat],
  result: Nat,
  body: (if (action =? 0) then 2 else (if (action =? 1) then 3 else (if (action =? 2) then 1 else (if (action =? 3) then 3 else (if (action =? 4) then 4 else (if (action =? 5) then 3 else targetStatus))))))
}

definition{
  name: nextPower,
  parameters: [action : Nat, targetPower : Nat, grantedPower : Nat],
  result: Nat,
  body: (if (action =? 7) then grantedPower else targetPower)
}

definition{
  name: applied,
  parameters: [permitted : Nat, action : Nat, targetStatus : Nat],
  result: Nat,
  body: (if (permitted =? 1) then (nextStatus action targetStatus) else targetStatus)
}

definition{
  name: statusAfter,
  parameters: [actions : (List Nat), start : Nat],
  result: Nat,
  body: (match actions
    | List.nil => start
    | List.cons head rest => (nextStatus head (statusAfter rest start))),
  recursive_argument: actions
}
\end{semanticmodule}
\end{lexlean}
