\begin{lexlean}{Comms.Sync}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\title{Boolean}
\begin{semanticmodule}
definition{
  name: holds,
  parameters: [store : (List Nat), kappa : Nat],
  result: Bool,
  body: (match store
    | List.nil => false
    | List.cons head rest => (if (head =? kappa) then true else (holds rest kappa))),
  recursive_argument: store
}

definition{
  name: wanted,
  parameters: [offered : (List Nat), store : (List Nat)],
  result: (List Nat),
  body: (match offered
    | List.nil => (nil Nat)
    | List.cons head rest => (if (holds store head) then (wanted rest store) else (cons head (wanted rest store)))),
  recursive_argument: offered
}

definition{
  name: bytesMove,
  parameters: [localHead : Nat, remoteHead : Nat],
  result: Bool,
  body: (if (localHead =? remoteHead) then false else true)
}

theorem{
  name: equalHeadsMoveNothing,
  parameters: [head : Nat],
  statement: ((bytesMove head head) = false),
  proof: linear_arithmetic{definitions: [bytesMove]},
  axioms: [propext]
}

theorem{
  name: emptyOfferWantsNothing,
  parameters: [store : (List Nat)],
  statement: ((wanted (nil Nat) store) = (nil Nat)),
  proof: simplify{definitions: [wanted]},
  axioms: [propext]
}

theorem{
  name: heldContentIsNotFetchedAgain,
  parameters: [kappa : Nat, rest : (List Nat), offered : (List Nat)],
  statement: ((wanted (cons kappa offered) (cons kappa rest)) = (wanted offered (cons kappa rest))),
  proof: linear_arithmetic{definitions: [holds, wanted]},
  axioms: [propext]
}
\end{semanticmodule}
\end{lexlean}
