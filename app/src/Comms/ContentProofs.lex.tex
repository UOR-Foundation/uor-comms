  name: nulInTextIsRefused,

  proof: induction{
    branches: [
      {
        proof: linear_arithmetic{definitions: [Comms.Content::badTextBytes, Comms.Event::joined]},
        binders: [],
        constructor: nil
      },
      {
        proof: linear_arithmetic{definitions: [Comms.Content::badTextBytes, Comms.Event::joined]},
        binders: [head, rest, ih],

      }

    scrutinee: front
  },
  axioms: [Classical.choice, Quot.sound, propext]
